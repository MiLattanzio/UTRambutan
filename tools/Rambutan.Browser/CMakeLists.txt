cmake_minimum_required(VERSION 3.22)
project(RambutanBrowser LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Qml Network WebEngineQuick)
find_package(X11 REQUIRED)
add_executable(rambutan-browser main.cpp browser.qrc)
target_link_libraries(rambutan-browser PRIVATE Qt6::Quick Qt6::Qml Qt6::Network Qt6::WebEngineQuick X11::X11)
add_executable(browser-policy-tests policy-tests.cpp)
add_executable(rambutan-browser-input-test main.cpp browser.qrc input-test.qrc)
target_compile_definitions(rambutan-browser-input-test PRIVATE BROWSER_INPUT_TEST)
target_link_libraries(rambutan-browser-input-test PRIVATE Qt6::Quick Qt6::Qml Qt6::Network Qt6::WebEngineQuick X11::X11)
target_compile_options(rambutan-browser-input-test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(browser-policy-tests PRIVATE Qt6::Core Qt6::Network)
target_compile_options(rambutan-browser PRIVATE -Wall -Wextra -Werror)
target_compile_options(browser-policy-tests PRIVATE -Wall -Wextra -Werror)
